
### Database Tables Required
```sql
-- Admin users table
CREATE TABLE admin_users (
  au_id INT PRIMARY KEY,
  au_username VARCHAR(50) UNIQUE,
  au_password_hash VARCHAR(255),
  au_full_name VARCHAR(100),
  au_email VARCHAR(100),
  au_role ENUM('ADMIN', 'EDITOR', 'VIEWER'),
  au_active CHAR(1),
  au_last_login DATETIME
);

-- Admin menu table
CREATE TABLE admin_menu (
  am_id INT PRIMARY KEY,
  am_menu_name VARCHAR(100),
  am_icon VARCHAR(50),
  am_controller VARCHAR(50),
  am_action VARCHAR(50),
  am_order_no INT,
  am_active CHAR(1),
  am_description VARCHAR(255)
);
```
